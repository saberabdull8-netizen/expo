export default function ExpiryCheckerApp() { const [image, setImage] = React.useState(null); const [result, setResult] = React.useState(""); const [loading, setLoading] = React.useState(false); const [detectedDate, setDetectedDate] = React.useState("");

const handleImageUpload = async (e) => { const file = e.target.files[0]; if (!file) return;

setImage(URL.createObjectURL(file));
setLoading(true);
setResult("");
setDetectedDate("");

try {
  const { createWorker } = await import("tesseract.js");
  const worker = await createWorker("eng");

  const {
    data: { text },
  } = await worker.recognize(file);

  await worker.terminate();

  const dateRegex = /(\d{2}[\/\-]\d{2}[\/\-]\d{4})|(\d{4}[\/\-]\d{2}[\/\-]\d{2})/;
  const match = text.match(dateRegex);

  if (!match) {
    setResult("ما قدرت أتعرف على التاريخ 😅");
    setLoading(false);
    return;
  }

  const foundDate = match[0];
  setDetectedDate(foundDate);

  let expiryDate;

  if (foundDate.includes("/")) {
    const parts = foundDate.split("/");

    if (parts[0].length === 4) {
      expiryDate = new Date(parts[0], parts[1] - 1, parts[2]);
    } else {
      expiryDate = new Date(parts[2], parts[1] - 1, parts[0]);
    }
  } else {
    const parts = foundDate.split("-");

    if (parts[0].length === 4) {
      expiryDate = new Date(parts[0], parts[1] - 1, parts[2]);
    } else {
      expiryDate = new Date(parts[2], parts[1] - 1, parts[0]);
    }
  }

  const today = new Date();

  if (expiryDate < today) {
    setResult("❌ المنتج منتهي الصلاحية");
  } else {
    setResult("✅ المنتج صالح وغير منتهي");
  }
} catch (error) {
  console.error(error);
  setResult("صار خطأ أثناء قراءة الصورة");
}

setLoading(false);

};

return ( <div className="min-h-screen bg-gray-100 flex items-center justify-center p-6"> <div className="bg-white shadow-xl rounded-3xl p-6 w-full max-w-md space-y-4"> <h1 className="text-2xl font-bold text-center"> فاحص تاريخ الصلاحية </h1>

<input
      type="file"
      accept="image/*"
      onChange={handleImageUpload}
      className="w-full border rounded-xl p-2"
    />

    {image && (
      <img
        src={image}
        alt="preview"
        className="rounded-2xl w-full h-64 object-cover border"
      />
    )}

    {loading && (
      <div className="text-center text-gray-600">جالس أقرأ التاريخ...</div>
    )}

    {detectedDate && (
      <div className="text-center text-lg font-semibold">
        التاريخ المكتشف: {detectedDate}
      </div>
    )}

    {result && (
      <div className="text-center text-xl font-bold p-3 rounded-2xl bg-gray-100">
        {result}
      </div>
    )}
  </div>
</div>

); }
